    Typing Rules of Capless #

    This defines the typing judgement C; Γ ⊢ t: E in Fig. 2, 5 and 6. Most rules correspond directly to the paper definitions, except for the Typed.bindc and Typed.bindt rules, which are for type and capture set bindings introduced in the mechanization.

    Note that the rules for boundary/break extension are also included in this definition.

    source
    inductive Capless.Typed {n m k : ℕ} :
    Context n m k → Term n m k → EType n m k → CaptureSet n k → Prop
    • var {n m k : ℕ} {Γ : Context n m k} {x : Fin n} {C : CaptureSet n k} {S : SType n m k} : (Γ.Bound x S ^ C) → Γ ⊢ Term.var x : EType.type S ^ {x=x} @ {x=x}
    • label {n m k : ℕ} {Γ : Context n m k} {x : Fin n} {S : SType n m k} : Γ.LBound x S → Γ ⊢ Term.var x : EType.type Label[S] ^ {x=x} @ {x=x}
    • pack {n m k : ℕ} {x : Fin n} {T : CType n m (k + 1)} {Γ : Context n m k} {C : CaptureSet n k} : ((Γ,c:=C) ⊢ Term.var x : EType.type T @ {x=x}) → Γ ⊢ Term.pack C x : ∃c.T @ ∅
    • sub {n m k : ℕ} {Γ : Context n m k} {t : Term n m k} {E1 : EType n m k} {C1 C2 : CaptureSet n k} {E2 : EType n m k} : (Γ ⊢ t : E1 @ C1) → (Γ ⊢ C1 <:c C2) → (Γ ⊢ E1 <:e E2) → Γ ⊢ t : E2 @ C2
    • abs {n k x✝ : ℕ} {Γ : Context n x✝ k} {T : CType n x✝ k} {t : Term (n + 1) x✝ k} {E : EType (n + 1) x✝ k} {C : CaptureSet n k} : ((Γ,x:T) ⊢ t : E @ C.weaken ∪ {x=0}) → Γ ⊢ (λ(x:T)t) : EType.type (∀(x:T)E) ^ C @ ∅
    • tabs {n k x✝ : ℕ} {Γ : Context n x✝ k} {S : SType n x✝ k} {t : Term n (x✝ + 1) k} {E : EType n (x✝ + 1) k} {C : CaptureSet n k} : ((Γ,X<:S) ⊢ t : E @ C) → Γ ⊢ (λ[X<:S]t) : EType.type (∀[X<:S]E) ^ C @ ∅
    • cabs {n k x✝ : ℕ} {Γ : Context n x✝ k} {B : CBound n k} {t : Term n x✝ (k + 1)} {E : EType n x✝ (k + 1)} {C : CaptureSet n k} : ((Γ,c<:B) ⊢ t : E @ C.cweaken) → Γ ⊢ (λ[c<:B]t) : EType.type (∀[c<:B]E) ^ C @ ∅
    • app {n m k : ℕ} {Γ : Context n m k} {x : Fin n} {C : CaptureSet n k} {T : CType n m k} {E : EType (n + 1) m k} {y : Fin n} : (Γ ⊢ Term.var x : EType.type (∀(x:T)E) ^ C @ {x=x}) → (Γ ⊢ Term.var y : EType.type T @ {x=y}) → Γ ⊢ Term.app x y : E.open y @ {x=x} ∪ {x=y}
    • invoke {n m k : ℕ} {Γ : Context n m k} {x : Fin n} {C : CaptureSet n k} {S : SType n m k} {y : Fin n} {E : EType n m k} : (Γ ⊢ Term.var x : EType.type Label[S] ^ C @ {x=x}) → (Γ ⊢ Term.var y : EType.type S ^ ∅ @ {x=y}) → Γ ⊢ Term.invoke x y : E @ {x=x} ∪ {x=y}
    • tapp {n m k : ℕ} {Γ : Context n m k} {x : Fin n} {C : CaptureSet n k} {X : Fin m} {E : EType n (m + 1) k} : (Γ ⊢ Term.var x : EType.type (∀[X<:SType.tvar X]E) ^ C @ {x=x}) → Γ ⊢ Term.tapp x X : E.topen X @ {x=x}
    • capp {n m k : ℕ} {Γ : Context n m k} {x : Fin n} {C : CaptureSet n k} {c : Fin k} {E : EType n m (k + 1)} : (Γ ⊢ Term.var x : EType.type (∀[c<:CBound.upper {c=c}]E) ^ C @ {x=x}) → Γ ⊢ Term.capp x c : E.copen c @ {x=x}
    • letin {n m k : ℕ} {Γ : Context n m k} {t : Term n m k} {T : CType n m k} {C : CaptureSet n k} {u : Term (n + 1) m k} {E : EType n m k} : (Γ ⊢ t : EType.type T @ C) → ((Γ,x:T) ⊢ u : E.weaken @ C.weaken) → Γ ⊢ (let x=t in u) : E @ C
    • letex {n m k : ℕ} {Γ : Context n m k} {t : Term n m k} {T : CType n m (k + 1)} {C : CaptureSet n k} {u : Term (n + 1) m (k + 1)} {E : EType n m k} : (Γ ⊢ t : ∃c.T @ C) → ((Γ,c<:*,x:T) ⊢ u : E.cweaken.weaken @ C.cweaken.weaken) → Γ ⊢ (let(c,x)=t in u) : E @ C
    • bindt {n m k : ℕ} {Γ : Context n m k} {S : SType n m k} {t : Term n (m + 1) k} {C : CaptureSet n k} {E : EType n m k} : ((Γ,X:=S) ⊢ t : E.tweaken @ C) → Γ ⊢ (let X=S in t) : E @ C
    • bindc {n m k : ℕ} {Γ : Context n m k} {C : CaptureSet n k} {t : Term n m (k + 1)} {E : EType n m k} {C0 : CaptureSet n k} : ((Γ,c:=C) ⊢ t : E.cweaken @ C0.cweaken) → Γ ⊢ (let c=C in t) : E @ C0
    • boundary {n m k : ℕ} {t : Term (n + 1) m (k + 1)} {C : CaptureSet n k} {Γ : Context n m k} {S : SType n m k} : ((Γ,c<:*,x:Label[S.cweaken] ^ {c=0}) ⊢ t : EType.type S.cweaken.weaken ^ ∅ @ C.cweaken.weaken ∪ {c=0} ∪ {x=0}) → Γ ⊢ (boundary:S in t) : EType.type S ^ CaptureSet.empty @ C
